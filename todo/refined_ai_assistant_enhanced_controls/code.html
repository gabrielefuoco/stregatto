def calculate_trajectory(v0, theta, g=9.81):
    """
    Calculates the trajectory of a projectile.
    """
    import math
    
    theta_rad = math.radians(theta)
    t_flight = (2 * v0 * math.sin(theta_rad)) / g
    
    return t_flight
